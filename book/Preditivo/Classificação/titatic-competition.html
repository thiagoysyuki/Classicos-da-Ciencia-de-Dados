from sklearn.linear_model import LinearRegression
from sklearn.metrics import r2_score, mean_squared_error

numeric_features = ['Age','SibSp','Parch','Fare']
categorical_features = ['Pclass','Embarked','Sex']


data_age = train[['Fare','SibSp','Parch','Age','Pclass','Sex','Embarked']].dropna()

preprocessor = ColumnTransformer(
    transformers=[
        ('cat', OneHotEncoder(), categorical_features)
    ]    
)

pipe_age = Pipeline(steps=[
    ('preprocess', preprocessor),
    ('regressor', RandomForestRegressor(max_depth=10, n_estimators=500))
]
)


X_age = data_age.drop('Age', axis=1)
y_age = data_age['Age']

pipe_age.fit(X_age,y_age)

y_pred = pipe_age.predict(X_age)
mse = mean_squared_error(y_age, y_pred)
r2 = r2_score(y_age, y_pred)
print(f'Mean Squared Error: {mse}')
